def trigger_elite_gemini_call(prompt_text):
    conversation_context = ""
    for msg in st.session_state.ai_chat_history[-6:]:
        if msg['content'] != prompt_text:
            conversation_context += f"{msg['role'].upper()}: {str(msg['content'])}\n\n"
            
    system_instruction = (
        "You are an expert dynamic context localized decision coach certified by the Society for Decision Professionals (SDP). "
        "Provide tailored responses, metrics framework guidelines, and structural options accordingly."
    )
    
    try:
        client = genai.Client(api_key=api_key)
        response = client.models.generate_content(
            model='gemini-3.6-flash',
            contents=f"{system_instruction}\n\nLOG:\n{conversation_context}\n\nUSER:\n{prompt_text}"
        )
        return response.text
    except Exception as e:
        return f"Communication Error: {str(e)}"
